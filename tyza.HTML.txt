script_js = r'''const packages = window.PACKAGES || {};
let selectedGame = null;
let selectedPackage = null;
let orderQty = 1;
let verifiedPlayerName = "";

const $ = (id) => document.getElementById(id);
const money = (n) => `Rs. ${Number(n || 0).toLocaleString("en-LK")}`;

function scrollToId(id) {
  document.getElementById(id)?.scrollIntoView({ behavior: "smooth", block: "start" });
}

function chooseGame(game) {
  selectedGame = game;
  selectedPackage = null;
  orderQty = 1;
  verifiedPlayerName = "";

  $("game-selection").classList.remove("hidden");
  $("store").classList.remove("hidden");
  $("order").classList.add("hidden");
  $("store-title").textContent = game;
  $("store-subtitle").textContent = game === "FREE FIRE"
    ? "Choose diamonds or subscription packages."
    : "Choose your Blood Strike package.";

  renderPackages();
  scrollToId("store");
}

function renderPackages() {
  const holder = $("package-sections");
  holder.innerHTML = "";

  const list = packages[selectedGame] || [];
  const groups = {};
  list.forEach(pkg => {
    const group = pkg.type === "subscription" ? "SUBSCRIPTION PACKAGES"
      : pkg.type === "diamond" ? "DIAMOND PACKAGES"
      : "BLOOD STRIKE PACKAGES";
    (groups[group] ||= []).push(pkg);
  });

  Object.entries(groups).forEach(([title, items]) => {
    const section = document.createElement("div");
    section.className = "package-group";
    section.innerHTML = `<h3>${title}</h3><div class="package-grid"></div>`;
    const grid = section.querySelector(".package-grid");

    items.forEach(pkg => {
      const card = document.createElement("article");
      card.className = "package-card";
      const visual = pkg.type === "diamond"
        ? `<img src="/static/assets/diamond-package.webp" alt="">`
        : pkg.type === "subscription"
          ? `<div class="package-icon">✦</div>`
          : pkg.type === "pass"
            ? `<div class="package-icon">PASS</div>`
            : `<div class="package-icon">✦</div>`;

      const unavailable = Number(pkg.price) === 0;
      card.innerHTML = `
        <div class="package-visual">${visual}</div>
        <div class="package-name">${escapeHtml(pkg.name)}</div>
        <div class="package-price">${unavailable ? "PRICE TO BE UPDATED" : money(pkg.price)}</div>
        <div class="package-bottom">
          <div class="quantity" data-qty="${pkg.id}">
            <button type="button" class="minus" aria-label="Decrease">−</button>
            <span>1</span>
            <button type="button" class="plus" aria-label="Increase">+</button>
          </div>
          <button type="button" class="select-btn" data-select="${pkg.id}" ${unavailable ? "disabled" : ""}>
            ${unavailable ? "COMING SOON" : "SELECT"}
          </button>
        </div>`;
      grid.appendChild(card);

      const q = card.querySelector(".quantity span");
      let localQty = 1;
      card.querySelector(".minus").addEventListener("click", () => {
        localQty = Math.max(1, localQty - 1);
        q.textContent = localQty;
      });
      card.querySelector(".plus").addEventListener("click", () => {
        localQty = Math.min(99, localQty + 1);
        q.textContent = localQty;
      });
      card.querySelector(".select-btn").addEventListener("click", () => {
        selectedPackage = pkg;
        orderQty = localQty;
        openOrder();
      });
    });

    holder.appendChild(section);
  });
}

function openOrder() {
  $("order").classList.remove("hidden");
  $("selected-game").textContent = selectedGame;
  $("selected-package").textContent = selectedPackage.name;
  $("summary-game").textContent = selectedGame;
  $("summary-package").textContent = selectedPackage.name;
  $("summary-qty").textContent = orderQty;
  $("order-quantity").textContent = orderQty;
  $("qty-value").textContent = orderQty;
  $("summary-total").textContent = money(selectedPackage.price * orderQty);
  $("player-name").textContent = "Not verified";
  $("lookup-status").textContent = "";
  verifiedPlayerName = "";
  scrollToId("order");
}

function updateOrderQty(delta) {
  orderQty = Math.max(1, Math.min(99, orderQty + delta));
  $("order-quantity").textContent = orderQty;
  $("qty-value").textContent = orderQty;
  $("summary-qty").textContent = orderQty;
  if (selectedPackage) $("summary-total").textContent = money(selectedPackage.price * orderQty);
}

async function checkPlayer() {
  const uid = $("uid").value.trim();
  const status = $("lookup-status");
  if (!uid) {
    status.textContent = "Enter your Player UID / ID first.";
    return;
  }

  status.textContent = "Checking player…";
  verifiedPlayerName = "";
  $("player-name").textContent = "Checking…";

  try {
    const response = await fetch("/lookup", {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({ game: selectedGame, uid })
    });
    const data = await response.json();

    if (data.ok && data.player_name) {
      verifiedPlayerName = data.player_name;
      $("player-name").textContent = data.player_name;
      status.textContent = "Player verified.";
    } else {
      $("player-name").textContent = "Not verified";
      status.textContent = data.message || "Player could not be verified.";
    }
  } catch (error) {
    $("player-name").textContent = "Not verified";
    status.textContent = "Player lookup service is unavailable right now.";
  }
}

$("check-player").addEventListener("click", checkPlayer);
$("order-minus").addEventListener("click", () => updateOrderQty(-1));
$("order-plus").addEventListener("click", () => updateOrderQty(1));

$("order-form").addEventListener("submit", (event) => {
  event.preventDefault();
  if (!selectedGame || !selectedPackage) return;

  const uid = $("uid").value.trim();
  const customer = $("customer-name").value.trim();
  const contact = $("contact").value.trim();

  if (!uid || !customer || !contact) {
    $("lookup-status").textContent = "Please complete all required fields.";
    return;
  }

  const playerLine = verifiedPlayerName ? `Player name: ${verifiedPlayerName}\n` : "";
  const total = selectedPackage.price * orderQty;
  const message = [
    "TYZA DIMAOND STORE — NEW ORDER",
    "",
    `Customer name: ${customer}`,
    `Contact: ${contact}`,
    `Game: ${selectedGame}`,
    `Player UID: ${uid}`,
    playerLine.trim(),
    `Package: ${selectedPackage.name}`,
    `Quantity: ${orderQty}`,
    `Price: ${money(selectedPackage.price)}`,
    `Total: ${money(total)}`
  ].filter(Boolean).join("\n");

  const url = `https://wa.me/94701014203?text=${encodeURIComponent(message)}`;
  window.open(url, "_blank", "noopener");
});

document.querySelectorAll(".game-card").forEach(card => {
  card.addEventListener("click", () => chooseGame(card.dataset.game));
});

$("change-game").addEventListener("click", () => {
  $("store").classList.add("hidden");
  $("order").classList.add("hidden");
  selectedGame = null;
  selectedPackage = null;
  scrollToId("game-selection");
});

function escapeHtml(value) {
  return String(value).replace(/[&<>"']/g, ch => ({
    "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#039;"
  }[ch]));
}
